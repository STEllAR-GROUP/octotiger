cmake_minimum_required(VERSION 3.20)
project(OctoTigerQb4HpxCheck LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(HPX 2.0 REQUIRED CONFIG NO_CMAKE_PACKAGE_REGISTRY)
if(NOT HPX_VERSION_MAJOR EQUAL 2)
  message(FATAL_ERROR "This branch requires the HPX 2 stack; selected ${HPX_VERSION_STRING}")
endif()
if(NOT HPX_WITH_NETWORKING OR NOT HPX_WITH_PARCELPORT_MPI)
  message(FATAL_ERROR "The selected HPX must have networking and its MPI parcelport enabled. Rebuild the matching HPX 2 dependency stack; consumer flags cannot enable an absent parcelport.")
endif()
find_package(MPI REQUIRED COMPONENTS CXX)
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/hpx-mpi-selection.txt"
  "HPX_DIR=${HPX_DIR}\nHPX_VERSION=${HPX_VERSION_STRING}\nHPX_GIT_COMMIT=${HPX_GIT_COMMIT}\nHPX_WITH_NETWORKING=${HPX_WITH_NETWORKING}\nHPX_WITH_PARCELPORT_MPI=${HPX_WITH_PARCELPORT_MPI}\nCXX_COMPILER=${CMAKE_CXX_COMPILER}\nMPI_CXX_COMPILER=${MPI_CXX_COMPILER}\nMPI_CXX_LIBRARIES=${MPI_CXX_LIBRARIES}\nMPIEXEC_EXECUTABLE=${MPIEXEC_EXECUTABLE}\n")
