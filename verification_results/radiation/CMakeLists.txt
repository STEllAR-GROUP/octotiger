cmake_minimum_required(VERSION 3.16)
project(radiation_results LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(OpenSSL REQUIRED)
find_package(nlohmann_json 3.9 REQUIRED)
find_path(SILO_INCLUDE_DIR silo.h HINTS ENV SILO_ROOT PATH_SUFFIXES include)
find_library(SILO_LIBRARY NAMES siloh5 silo HINTS ENV SILO_ROOT PATH_SUFFIXES lib lib64)
if(NOT SILO_INCLUDE_DIR OR NOT SILO_LIBRARY)
  message(FATAL_ERROR "Silo required. Set SILO_ROOT or SILO_INCLUDE_DIR and SILO_LIBRARY.")
endif()
set(RR_SOURCES cpp/common.cpp cpp/data.cpp cpp/plot.cpp cpp/movie.cpp cpp/visit_session.cpp cpp/run.cpp cpp/maintenance.cpp cpp/report.cpp)
add_library(rr_core STATIC ${RR_SOURCES})
target_include_directories(rr_core PUBLIC cpp ${SILO_INCLUDE_DIR})
target_link_libraries(rr_core PUBLIC nlohmann_json::nlohmann_json OpenSSL::Crypto ${SILO_LIBRARY})
target_compile_options(rr_core PRIVATE -Wall -Wextra -Wpedantic)
add_executable(radiation-results cpp/main.cpp)
target_link_libraries(radiation-results PRIVATE rr_core)
get_filename_component(DEFAULT_OCTOTIGER_SOURCE_DIR "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
set(OCTOTIGER_SOURCE_DIR "${DEFAULT_OCTOTIGER_SOURCE_DIR}" CACHE PATH "Octo-TIGER source checkout")
if(NOT EXISTS "${OCTOTIGER_SOURCE_DIR}/octotiger/test_problems/radiation/reference.hpp")
  message(FATAL_ERROR "Missing Octo-TIGER radiation reference headers below the repository root. Set OCTOTIGER_SOURCE_DIR to the checkout.")
endif()
find_path(FFTW3_INCLUDE_DIR fftw3.h)
find_library(FFTW3_LIBRARY NAMES fftw3)
if(NOT FFTW3_INCLUDE_DIR OR NOT FFTW3_LIBRARY)
  message(FATAL_ERROR "The Gaussian reference generator needs FFTW3. Install libfftw3-dev.")
endif()
add_executable(gen_radiation_reference cpp/gen_radiation_reference.cpp)
target_compile_features(gen_radiation_reference PRIVATE cxx_std_23)
target_include_directories(gen_radiation_reference PRIVATE "${OCTOTIGER_SOURCE_DIR}" "${FFTW3_INCLUDE_DIR}")
target_link_libraries(gen_radiation_reference PRIVATE "${FFTW3_LIBRARY}")
add_dependencies(radiation-results gen_radiation_reference)
include(CTest)
if(BUILD_TESTING)
  add_executable(rr-tests tests_cpp/tests.cpp)
  target_link_libraries(rr-tests PRIVATE rr_core)
  add_test(NAME radiation-results-readers COMMAND rr-tests)
  find_program(GNUPLOT_EXECUTABLE NAMES gnuplot gnuplot-nox)
  if(GNUPLOT_EXECUTABLE)
    add_executable(rr-integration tests_cpp/integration.cpp)
    target_link_libraries(rr-integration PRIVATE rr_core)
    add_test(NAME radiation-results-resume
      COMMAND rr-integration $<TARGET_FILE:radiation-results> $<TARGET_FILE:rr-tests>
              ${CMAKE_CURRENT_SOURCE_DIR} ${GNUPLOT_EXECUTABLE} ${CMAKE_COMMAND})
    find_program(FFMPEG_EXECUTABLE NAMES ffmpeg)
    if(FFMPEG_EXECUTABLE)
      add_executable(rr-terminal tests_cpp/terminal.cpp)
      target_link_libraries(rr-terminal PRIVATE rr_core util)
      add_test(NAME radiation-results-terminal
        COMMAND rr-terminal $<TARGET_FILE:radiation-results> $<TARGET_FILE:rr-tests>
                ${GNUPLOT_EXECUTABLE} ${FFMPEG_EXECUTABLE})
      set_tests_properties(radiation-results-terminal PROPERTIES TIMEOUT 40)
    endif()
  endif()
endif()
